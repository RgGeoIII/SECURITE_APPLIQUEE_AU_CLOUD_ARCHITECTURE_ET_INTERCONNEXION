# Bastion
aws ec2 run-instances --region eu-west-3 \
  --image-id ami-05dfcc4b49790367c \
  --instance-type t2.micro \
  --key-name cle-td-geogeo \
  --subnet-id subnet-0b262aa3958b5b1b6 \
  --associate-public-ip-address \
  --tag-specifications "ResourceType=instance,Tags=[{Key=Name,Value=geogeobastion}]" \
  --count 1 \
  --query "Instances[0].InstanceId"

# Cible
aws ec2 run-instances --region eu-west-3 \
  --image-id ami-05dfcc4b49790367c \
  --instance-type t2.micro \
  --key-name cle-td-geogeo \
  --subnet-id subnet-0b262aa3958b5b1b6 \
  --no-associate-public-ip-address \
  --tag-specifications "ResourceType=instance,Tags=[{Key=Name,Value=geogeocible}]" \
  --count 1 \
  --query "Instances[0].InstanceId"
